cmake_minimum_required(VERSION 3.16)
project(passport_auth_ui_render C)
set(CMAKE_C_STANDARD 11)
get_filename_component(PASSPORT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
add_subdirectory("${PASSPORT_ROOT}/managed_components/lvgl__lvgl" "${CMAKE_BINARY_DIR}/lvgl")
add_executable(auth_ui_render_tests "${PASSPORT_ROOT}/tests/test_auth_ui_render.c")
target_compile_definitions(auth_ui_render_tests PRIVATE ESP_PLATFORM _POSIX_C_SOURCE=200809L)
target_include_directories(auth_ui_render_tests PRIVATE "${PASSPORT_ROOT}/main")
target_link_libraries(auth_ui_render_tests PRIVATE lvgl m)
enable_testing()
add_test(NAME auth_ui_render_pressure COMMAND auth_ui_render_tests 15728)
